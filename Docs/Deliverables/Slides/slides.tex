% 3 ---------------- problem + architecture
\begin{frame}[c]{1. Problem, objectives and architecture}
\begin{columns}[T]
\column{0.53\textwidth}
\textbf{Can a complete, playable 3D game be made by code alone on a student laptop?}
\vspace{3pt}
\begin{block}{Two hard limits}
\begin{itemize}\setlength{\itemsep}{1pt}
 \item No model, image or precomputed data file
 \item macOS OpenGL 4.1: no compute shaders
\end{itemize}
\end{block}
\vspace{2pt}
\textbf{Objectives}
\begin{itemize}\small\setlength{\itemsep}{0pt}
 \item Layered engine, downward dependencies
 \item All content from code and seeds
 \item Modern renderer, playable game
 \item Verified: tests, 16.7\,ms budget
\end{itemize}

% 4 ---------------- procedural
\begin{frame}[c]{2. Procedural shapes and textures}
\centering\pic{0.62}{noise_stages.png}\\[-1pt]{\small gradient, fractal, warped, ridged and cellular noise}
\vspace{4pt}
\begin{columns}[T]
\column{0.37\textwidth}
\begin{block}{Fractal noise (tiles exactly)}
\small$\displaystyle F(\mathbf x)=\frac{\sum_{i=0}^{n-1}2^{-i}\nu(2^i\mathbf x)}{\sum_{i=0}^{n-1}2^{-i}}$
\end{block}
\begin{block}{Vertex push}
\small$\mathbf p'=\mathbf p+A\,\nu(f\mathbf p)\,\hat n,\ |\mathbf p'-\mathbf p|\le A$
\end{block}
\column{0.6\textwidth}
\begin{itemize}\setlength{\itemsep}{2pt}
 \item 18 recipes $\to$ 22 materials, baked once at 1024$^2$
 \item \textbf{Scharr} filter turns height into bumps; crevices darken
 \item Objects = tables of primitives + noise deformers
 \item Soldiers: skeletons with two-bone IK legs
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}[c]{3. The rendering pipeline}
\centering
\begin{tikzpicture}[node distance=0.15cm and 0.22cm]
\node[box] (s) {Sky\\table};\node[box,right=of s] (sh) {Shadows\\3 cascades};\node[box,right=of sh] (g) {Geometry\\G-buffer};\node[box,right=of g] (ao) {SSAO};\node[box,right=of ao] (li) {Lighting,\\reflect, fog};\node[box,right=of li] (pa) {Smoke,\\fire};\node[box,right=of pa] (po) {Bloom, ACES,\\FXAA};
\foreach \x/\y in {s/sh,sh/g,g/ao,ao/li,li/pa,pa/po}\draw[arr](\x)--(\y);
\end{tikzpicture}
\vspace{4pt}
\begin{columns}[T]
\column{0.56\textwidth}\raggedright
\textbf{Deferred shading:} store each pixel's colour, normal, shininess and depth first, then light it once.
\begin{block}{Cook--Torrance reflection}
\small$f=\frac{(1-F_l)(1-F_v)\mathbf c_d}{\pi}+DVF$\\[1pt]
\footnotesize$D=\frac{\alpha^2}{\pi((\mathbf n\cdot\mathbf h)^2(\alpha^2-1)+1)^2}$, $F=F_0+(1-F_0)(1-\mathbf v\cdot\mathbf h)^5$
\end{block}
\begin{block}{Fog and tone map}
\small$C=TC_s+(1-T)C_h,\ T=e^{-kd}$\hfill $C_{\rm out}=\frac{x(2.51x+0.03)}{x(2.43x+0.59)+0.14}$
\end{block}
\column{0.42\textwidth}
\pic{1}{dusk.jpg}\\[1pt]{\small\centering Dusk: sun, sky, haze, shadows}
\end{columns}
\end{frame}

 \item \textbf{Mission:} hack, destroy, rescue, extract; stealth AI; synthesised sound

% 8 ---------------- challenge
\begin{frame}[c]{6. Challenge solved, limits and future work}
\textbf{Bug: a bright vertical line at night.} The fog colour sampled the \emph{whole} sky, so one horizon star painted a streak on every surface in its column. Fix: fog uses the smooth atmosphere only.
\vspace{3pt}
\begin{columns}[T]
\column{0.27\textwidth}\pic{1}{gate_night_bug.jpg}\\[-2pt]{\footnotesize\centering before}\\[1pt]\pic{1}{gate_night.jpg}\\[-2pt]{\footnotesize\centering after}
\column{0.69\textwidth}
\begin{block}{Limitations}\small Blocky primitive models; no true global illumination; lumpy foliage; tuned by scripts and screenshots.\end{block}
\begin{block}{Future work}\small Leaf cards, screen-space GI, point-light shadows, multiplayer, a compute-shader port.\end{block}
\begin{block}{Take-away}\small Most ``art'' is a few simple ideas, applied consistently and tested.\end{block}
\end{columns}
