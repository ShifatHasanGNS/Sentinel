% 3 ---------------- problem + architecture
\begin{frame}[c]{1. Problem, objectives and architecture}
\begin{columns}[T]
\column{0.55\textwidth}
\textbf{Can a complete, playable 3D game be made by code alone on a student laptop?}
\vspace{2pt}
\begin{block}{Two hard limits}\small
No model, image or precomputed file; OpenGL 4.1, no compute shaders.
\end{block}
\begin{block}{Objectives}\small
Layered engine; all content from code; modern renderer; a playable game (vehicles, ladders, stealth, five-objective mission); tested, 16.7\,ms budget.
\end{block}
\column{0.42\textwidth}
\centering
