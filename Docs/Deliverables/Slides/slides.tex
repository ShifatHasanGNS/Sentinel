\begin{frame}[c]{1. Problem, objectives and architecture}
\begin{columns}[T]
\column{0.55\textwidth}
\textbf{Can a complete, playable 3D game be made by code alone on a student laptop?}
\vspace{2pt}
\begin{block}{Two hard limits}\small
No model, image or precomputed file; OpenGL 4.1, no compute shaders.
\end{block}
\begin{block}{Objectives}\small
Layered engine; all content from code; modern renderer; playable game; tested, 16.7\,ms budget.
\end{block}
\begin{block}{The game}\small
Drivable vehicles and helicopter; ladders, doors, stealth enemies; five-objective mission.
\end{block}

% 4 ---------------- procedural
\begin{frame}[c]{2. Procedural shapes and textures}
\centering\pic{0.6}{noise_stages.png}\\[-1pt]{\small gradient, fractal, warped, ridged and cellular noise}
\vspace{4pt}
\begin{columns}[T]
\column{0.4\textwidth}
\begin{block}{Fractal noise (tiles exactly)}
\small$\displaystyle F(\mathbf x)=\frac{\sum_{i=0}^{n-1}2^{-i}\nu(2^i\mathbf x)}{\sum_{i=0}^{n-1}2^{-i}}$
\end{block}
\begin{block}{Vertex push}
\small$\mathbf p'=\mathbf p+A\,\nu(f\mathbf p)\,\hat n,\ |\mathbf p'-\mathbf p|\le A$
\end{block}
\column{0.57\textwidth}
\begin{itemize}\setlength{\itemsep}{2pt}
 \item 18 recipes $\to$ 22 materials, baked once at 1024$^2$
 \item \textbf{Scharr} filter turns height into bumps; crevices darken
 \item Objects: tables of primitives and noise deformers
 \item Soldiers: skeletons with two-bone IK legs
\end{itemize}
\end{columns}
\end{frame}

% 5 ---------------- transformations
\begin{frame}[c]{3. Transformations: creating and combining objects}
\centering
\begin{tikzpicture}[node distance=0.12cm and 0.2cm]
\node[box] (a) {Primitive\\mesh};\node[box,right=of a] (b) {Deformers\\own space};\node[box,right=of b] (c) {$M_{\rm part}$\\scale, turn, move};\node[box,right=of c] (d) {Merge by\\material};\node[box,right=of d] (e) {$M_{\rm world}$\\instances};\node[box,right=of e] (f) {$V$ then $P$\\to screen};
\foreach \x/\y in {a/b,b/c,c/d,d/e,e/f}\draw[arr](\x)--(\y);
\end{tikzpicture}
\vspace{4pt}
\begin{columns}[T]
\column{0.52\textwidth}\raggedright
\begin{block}{One part: scale, turn, then move}
$M_{\rm part}=T(\mathbf t)\,R_z R_y R_x\,S(\mathbf s)$\\[1pt]
\small Right to left: stretched, turned about its own centre, moved last. $(1,0,0)\to S(2,1,1)\to R_y90^\circ\to T(5,0,0)=(5,0,-2)$
\end{block}
\begin{block}{Normals: inverse transpose}
$\hat n'=\dfrac{(L^{-1})^{\top}\hat n}{\lVert(L^{-1})^{\top}\hat n\rVert}$\\[1pt]
\small $S=\mathrm{diag}(2,1,1)$: correct $(0.447,0.894,0)$, naive $(0.894,0.447,0)$
\end{block}
\column{0.45\textwidth}
\begin{block}{Rotation about $y$}
\small$R_y(\theta)=\begin{pmatrix}\cos\theta&0&\sin\theta\\0&1&0\\-\sin\theta&0&\cos\theta\end{pmatrix}$
\end{block}
\begin{block}{Hierarchy and screen}
\small$M_{\rm gun}=M_{\rm hull}\,T(\mathbf c_t)R_y(\theta_t)\,T(\mathbf c_g)R_x(-\theta_g)$\\[2pt]
$\mathbf p_{\rm clip}=P\,V\,M_{\rm world}\,M_{\rm part}\,\mathbf p$
\end{block}
\end{columns}
\end{frame}

% 6 ---------------- illumination
\begin{frame}[c]{4. Illumination models and light sources}
\centering{\small$\displaystyle L_o=\sum_j f(\mathbf l_j,\mathbf v)\,L_j\,\mathrm{att}_j\,\mathrm{sh}_j\max(0,\mathbf n\!\cdot\!\mathbf l_j)+\mathrm{ambient}\cdot\mathrm{AO}+\mathbf e$}
\vspace{3pt}
\begin{columns}[T]
\column{0.63\textwidth}\small
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{@{}p{2.5cm}l@{}}
\textbf{Lambert} & $\frac{\mathbf c_d}{\pi}$\\
\textbf{Phong} & $\frac{\mathbf c_d(1-F_0)}{\pi}+F_0\frac{m+2}{2\pi}(\mathbf r\!\cdot\!\mathbf v)^m$\\
\textbf{Blinn--Phong} & $\frac{\mathbf c_d(1-F_0)}{\pi}+F_0\frac{m+8}{8\pi}(\mathbf n\!\cdot\!\mathbf h)^m$\\
\textbf{Oren--Nayar} & $\frac{\mathbf c_d}{\pi}(A+B\max(0,\cos\Delta\varphi)\sin\alpha\tan\beta)$\\
\textbf{Cook--Torrance} & $\frac{(1-F_l)(1-F_v)\mathbf c_d}{\pi}+DVF$\\
\textbf{Subsurface} & cosine $\to\frac{\mathbf n\cdot\mathbf l+w}{1+w}$, $w=0.5$
\end{tabular}
\column{0.35\textwidth}
\begin{block}{Light sources}
\normalsize Directional: sun, moon\\ Point: lamps, flashes\\ Spot: floodlights (cone)\\ Area: $4\times4$ samples\\[2pt]
\small $\mathrm{att}=\frac{\mathrm{clamp}(1-(d/R)^4)^2}{d^2+1}$
\end{block}
\pic{0.95}{models.jpg}\\[1pt]{\scriptsize\centering Six models, one sphere each}
\end{columns}
\end{frame}

% 7 ---------------- rendering
\begin{frame}[c]{5. The rendering pipeline: shadows, ambient, tone map}
\centering
\begin{tikzpicture}[node distance=0.15cm and 0.22cm]
\node[box] (s) {Sky\\table};\node[box,right=of s] (sh) {Shadows\\3 cascades};\node[box,right=of sh] (g) {Geometry\\G-buffer};\node[box,right=of g] (ao) {SSAO};\node[box,right=of ao] (li) {Lighting,\\reflect, fog};\node[box,right=of li] (pa) {Smoke,\\fire};\node[box,right=of pa] (po) {Bloom, ACES,\\FXAA};
\foreach \x/\y in {s/sh,sh/g,g/ao,ao/li,li/pa,pa/po}\draw[arr](\x)--(\y);
\end{tikzpicture}
\vspace{3pt}
\begin{columns}[T]
\column{0.56\textwidth}\raggedright
\textbf{Deferred shading:} store colour, normal, shininess, depth first; light each pixel once.
\begin{block}{Cascaded shadow maps}
\footnotesize$z_i=\lambda n(\frac{r}{n})^{i/N}+(1-\lambda)(n+(r-n)\frac iN)$, $\lambda=0.75$, $N=3$: cuts at 9.1, 24.2, 100\,m; normal-offset lookup, 8 rotated PCF taps.
\end{block}
\begin{block}{Fog, tone map, ambient}
\footnotesize$C=TC_s+(1-T)C_h,\ T=e^{-kd}$\hfill $C_{\rm out}=\frac{x(2.51x+0.03)}{x(2.43x+0.59)+0.14}$\\ ambient = sky diffuse + split-sum specular, times SSAO.
\end{block}
\column{0.42\textwidth}
\pic{1}{dusk.jpg}\\[1pt]{\small\centering Dusk: sun, sky, haze, shadows}
\end{columns}
\end{frame}
